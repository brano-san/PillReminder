/**
 * Сколько приёмов набора показать как выпитые. Считаем внутри текущего набора (во втором цикле суток
 * «4 из 3» выглядело бы дико), но закрытый набор показываем как «N из N», а не «0 из N».
 */
fun takenInSet(taken: Int, total: Int, hasPending: Boolean): Int {
    val size = total.coerceAtLeast(1)
    val inSet = taken % size
    return if (inSet == 0 && taken > 0 && !hasPending) size else inSet
}
```

// data/MealGate.kt
/** Сколько ждать еду после базового времени приёма «после еды», прежде чем напомнить без неё. */
const val MEAL_WAIT_MAX_MS = 3 * 60 * 60_000L

/**
 * Приём «после еды» ждёт кнопку «Поел»: еда должна быть отмечена после порога — подъёма дня для первого
 * приёма набора, иначе фактического времени предыдущего приёма этой таблетки в тот же день.
 * «По часам» и приёмы без правила не ждут. Нет данных о подъёме — не ждём (нечего сравнивать).
 */
fun mealSatisfied(med: Medication, dose: Dose, dayDoses: List<Dose>, meals: List<Long>, wakeAt: Long?): Boolean {
    if (med.afterMealMinutes <= 0 || med.byClock) return true
    val previous = dayDoses
        .filter { it.medId == dose.medId && it.dayEpochDay == dose.dayEpochDay && it.indexInDay < dose.indexInDay && it.status != DoseStatus.PENDING }
        .maxByOrNull { it.indexInDay }
    val threshold = previous?.let { it.takenAt ?: it.plannedAt } ?: wakeAt ?: return true
    return meals.any { it >= threshold }
}

// data/Dao.kt, MealDao
@Query("SELECT atMillis FROM meals ORDER BY atMillis")
fun observeAllTimes(): Flow<List<Long>>

// ui/MainViewModel.kt
data class MedRow(..., val waitsMeal: Boolean = false)
data class HomeState(..., val bedAt: Long? = null, val doses: List<Dose> = emptyList(), val meals: List<Long> = emptyList())

// ui/Lang.kt
val waitsMealShort: String  // RU «ждёт «Поел»» / EN "waits for \"Ate\""
```
